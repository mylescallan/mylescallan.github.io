var today = new Date();         <- Uses Global Object "Date"
var hourNow = today.getHours(); <- "today" inherits the properties of "Date"
var greeting;

if (hourNow > 18) {             <- Conditional Statement
    greeting = 'Good evening!';
} else if (hourNow > 12) {
    greeting = 'Good afternoon!';
} else if (hourNow > 0) {
    greeting = 'Good morning!';
} else {
    greeting = 'Welcome!';
}

document.write('<h3>' + greeting + '</h3>');  <- Uses Global Object
                                                        "document" and its
                                                        "write" method

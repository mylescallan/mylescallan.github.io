Chapter 4


This chapter introduces loops, and global objects. In this page, the code generates an addition table (and again, using CSS, places it over a background image).

Code for Chapter 4

var table = 3;             // Unit of table
var operator = 'other';    // Type of calculation (if 'addition' then addition,
                           //    else, multiplication)
var i = 1;                 // Set counter to 1
var msg = '';              // Message (which is the array created below
                           // adding a new row each loop (using the shorthand "+=")

if (operator === 'addition') {
  // Do addition
  while (i < 11) {
    msg += i + ' + ' + table + ' = ' + (i + table) + '
                        
';
    i++;
  }
} else {
  // Do multiplication
  while (i < 11) {
    msg += i + ' x ' + table + ' = ' + (i * table) + '
                        
';
    i++;
  }
}

// Write the message into the page
var el = document.getElementById('blackboard');
el.innerHTML = msg;

By changing the var "operator" to anything other than 'addition' you get the multiplication table (as I did here).
